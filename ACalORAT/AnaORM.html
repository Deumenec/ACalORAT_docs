101    def dRij_dEnergy(self, fringes=True):
102        """
103        Analytical derivative of the ORM with respect to the beam energy
104        deviation, dR/ddelta, from the base optics only (no RF-shifted rings).
105
106        Chain rule over the chromatic gradient sources:
107          - quadrupoles and CFDs: dK1/ddelta = -PolynomB[1] (the effective
108            gradient scales as K1/(1+delta)), thick formula dRij_dqk_thick23
109          - dipole edge focusing at the CFD faces (thin fringe quadrupoles,
110            K1L_edge = -h*tan(E) scaling as 1/(1+delta)), thin formula
111            dRij_dqk_thick2 with the local face optics
112          - sextupoles: feed-down from the dispersion orbit eta*ddelta,
113            integrated exactly along each sextupole (dRi_dk_sex_term with
114            i = eta, j = eta', k = 0)
115
116        Returns shape (n_BPM, n_cor).  Uses private Elements instances, so the
117        broadcasting state of self.bpm/self.cor/... is not modified.
118        """
119        bpm = Elements(self.ring, self.all_optics, self.ind["bpm"], self.dir_ind, self.sgn)
120        cor = Elements(self.ring, self.all_optics, self.ind["cor"][self.dir], self.dir_ind, self.sgn)
121        bpm.broadcasters(1, 3)
122        cor.broadcasters(2, 3)
123
124        out = 0.0
125        # thick gradients: quads and CFDs
126        for key in ("quad", "CFD"):
127            if key not in self.ind or len(self.ind[key]) == 0:
128                continue
129            el = Elements(self.ring, self.all_optics, self.ind[key], self.dir_ind, self.sgn)
130            el.broadcasters(0, 3)
131            w = -np.array([self.ring[i].PolynomB[1] for i in self.ind[key]])
132            out = out + self.dRij_dqk_thick23_wsum(bpm, cor, el, w)
133
134        # Thin fringe quadrupoles at the CFD faces, horizontal plane only:
135        # empirically the vertical edge-chromaticity contribution cancels in the
136        # AT edge model (adding it degrades the vertical dR/ddelta), while the
137        # horizontal one improves the agreement substantially.
138        # TODO: derive the vertical cancellation from the AT edge_fringe map.
139        if fringes and self.dir_ind == 0 and "CFD" in self.ind:
140            cfd_ind = np.asarray(self.ind["CFD"])
141            h = np.array([getattr(self.ring[i], "BendingAngle", 0.0) / self.ring[i].Length
142                          for i in cfd_ind])
143            for face_ind, w_face in [
144                (cfd_ind,     h * np.tan(np.array([getattr(self.ring[i], "EntranceAngle", 0.0) for i in cfd_ind]))),
145                (cfd_ind + 1, h * np.tan(np.array([getattr(self.ring[i], "ExitAngle",     0.0) for i in cfd_ind]))),
146            ]:
147                face = Elements(self.ring, self.all_optics, face_ind, self.dir_ind, self.sgn)
148                face.broadcasters(0, 3)
149                out = out + self.dRij_dqk_thick2_wsum(bpm, cor, face, w_face)
150
151        # sextupole feed-down from the dispersion orbit
152        if "sex" in self.ind and len(self.ind["sex"]) > 0:
153            bpm4 = Elements(self.ring, self.all_optics, self.ind["bpm"], self.dir_ind, self.sgn)
154            cor4 = Elements(self.ring, self.all_optics, self.ind["cor"][self.dir], self.dir_ind, self.sgn)
155            sex4 = Elements(self.ring, self.all_optics, self.ind["sex"], self.dir_ind, self.sgn)
156            bpm4.broadcasters(1, 4)
157            cor4.broadcasters(2, 4)
158            sex4.broadcasters(3, 4)
159            # horizontal dispersion at the sextupole entrances (feed-down source in both planes)
160            eta  = np.array([self.all_optics[2]["dispersion"][i][0] for i in self.ind["sex"]])
161            etap = np.array([self.all_optics[2]["dispersion"][i][1] for i in self.ind["sex"]])
162            ii = eta[None, None, None, :]
163            jj = etap[None, None, None, :]
164            kk = np.zeros_like(ii)
165            out = out + np.squeeze(self.dRi_dk_sex_term(bpm4, cor4, bpm4, sex4, ii, jj, kk))
166
167        return np.real(out)

173    def _private_element(self, key):
174        """Fresh Elements for `self.ind[key]` (or a list of indices), in this
175        plane, with no broadcasting configured — so the shared self.bpm/... are
176        never mutated by the assembly helpers."""

180    def _sex_dtheta(self, kicks_h):
181        """Map the horizontal feedback kicks (n_pert, n_cor_h) to the kick of the
182        corrector coil housed inside each sextupole (n_pert, n_sex), zero where the
183        sextupole has no corrector coil.  The feed-down orbit is horizontal, so the
184        relevant coils are always the horizontal correctors regardless of plane."""
185        hcm_col = {int(m): c for c, m in enumerate(self.ind["cor"]["h"])}
186        out = np.zeros((kicks_h.shape[0], len(self.ind["sex"])))
187        for s_i, l in enumerate(self.ind["sex"]):
188            c = hcm_col.get(int(l))
189            if c is not None:
190                out[:, s_i] = kicks_h[:, c]
191        return out

193    def dRij_dCFD_sex_term(self, x_sex, xp_sex, kicks_h, energy):
194        """
195        Exact sextupole feed-down contribution to the CFD ORM Jacobian in this
196        plane (Eqs. 99-101 of ACDIV-2017-01), assembled from the feedback orbit
197        at the sextupoles.
198
199        The orbit that drives the feed-down is the *corrector-driven* part only:
200        the dispersion orbit eta*delta is subtracted because its feed-down is
201        already carried by the energy term dR/ddelta (with sextupoles ON this is
202        the chromaticity correction itself; keeping it in both double-counts).
203
204        Parameters
205        ----------
206        x_sex, xp_sex : (n_pert, n_sex)
207            Horizontal orbit and slope at the sextupole entrances per unit CFD
208            change (numerical feedback results or their analytical estimates).
209        kicks_h : (n_pert, n_cor_h)
210            Horizontal feedback kicks per unit CFD change.
211        energy : (n_pert,)
212            Energy shift dp/p per unit CFD change applied by the RF feedback.
213
214        Returns shape (n_pert, n_BPM, n_cor).
215        """
216        eta_s  = np.array([self.all_optics[2]["dispersion"][i][0] for i in self.ind["sex"]])
217        etap_s = np.array([self.all_optics[2]["dispersion"][i][1] for i in self.ind["sex"]])
218        dtheta = self._sex_dtheta(kicks_h)
219
220        i_fd = (x_sex  - np.outer(energy, eta_s ))[:, None, None, :]
221        j_fd = (xp_sex - np.outer(energy, etap_s))[:, None, None, :]
222        k_fd = dtheta[:, None, None, :]
223
224        bpm = self._private_element("bpm")
225        cor = self._private_element(self.ind["cor"][self.dir])
226        sex = self._private_element("sex")
227        bpm.broadcasters(1, 4)
228        cor.broadcasters(2, 4)
229        sex.broadcasters(3, 4)
230        # Ek is unused inside dRi_dk_sex_term (kept for signature symmetry) -> pass bpm.
231        return self.dRi_dk_sex_term(bpm, cor, bpm, sex, i_fd, j_fd, k_fd)

233    def feedback_ingredients(self):
234        """
235        Analytical estimates of the feedback quantities needed to assemble the CFD
236        ORM (and dispersion) Jacobians, all from the single base-optics call:
237
238          - energy : (n_CFD,)          energy shift dp/p per unit CFD change
239          - kicks_h: (n_CFD, n_cor_h)  horizontal feedback kicks per unit CFD change
240          - x_sex  : (n_CFD, n_sex)    horizontal orbit at the sextupole entrances
241          - xp_sex : (n_CFD, n_sex)    horizontal slope at the sextupole entrances
242
243        These reproduce the numerical Cor_SVD feedback results (energy via
244        `dCFD_denergy`, kicks via `dkicksdCFDk`, orbit via `dxldCFDk`, slope by a
245        finite difference of the orbit across the field-free drift upstream of each
246        sextupole — exact because the orbit is a straight line in a drift).
247
248        Must be called on a *horizontal* AnaORM (the Cor_SVD feedback is horizontal).
249        Returns a dict ready to pass to `dRij_dCFD`.
250        """
251        # energy shift per CFD
252        bpm, cor, cfd = self._private_element("bpm"), self._private_element(self.ind["cor"][self.dir]), self._private_element("CFD")
253        bpm.broadcasters(0, 2); cor.broadcasters(1, 2); cfd.broadcasters(1, 2)
254        energy = np.real(self.dCFD_denergy(bpm, cor, cfd))
255
256        # horizontal feedback kicks per CFD
257        bpm, cor, cfd = self._private_element("bpm"), self._private_element(self.ind["cor"][self.dir]), self._private_element("CFD")
258        bpm.broadcasters(0, 3); cor.broadcasters(1, 3); cfd.broadcasters(2, 3)
259        kicks_h = np.real(self.dkicksdCFDk(bpm, cor, cfd))
260
261        # orbit and slope at the sextupoles (slope via the upstream drift)
262        up, L_drift = [], []
263        for l in self.ind["sex"]:
264            j = int(l) - 1
265            while self.ring[j].Length == 0:
266                j -= 1
267            up.append(j)
268            L_drift.append(self.ring[j].Length)
269        L_drift = np.array(L_drift)
270
271        bpm, cor, cfd = self._private_element("bpm"), self._private_element(self.ind["cor"][self.dir]), self._private_element("CFD")
272        sex, sexup = self._private_element("sex"), self._private_element(up)
273        bpm.broadcasters(0, 4); cor.broadcasters(1, 4); cfd.broadcasters(2, 4)
274        sex.broadcasters(3, 4); sexup.broadcasters(3, 4)
275        x_sex = np.real(self.dxldCFDk(bpm, cor, cfd, sex))
276        x_up  = np.real(self.dxldCFDk(bpm, cor, cfd, sexup))
277        xp_sex = (x_sex - x_up) / L_drift
278
279        return {"energy": energy, "kicks_h": kicks_h, "x_sex": x_sex, "xp_sex": xp_sex}

281    def dRij_dCFD(self, energy, kicks_h, x_sex, xp_sex, dRij_dE=None):
282        """
283        Full ORM Jacobian dR/dK_CFD in this plane under active Cor_SVD orbit + RF
284        feedback:
285
286            dR/dK = quadrupole (K1) term
287                  + dispersion-term derivatives          (horizontal only)
288                  + energy term  delta_k * dR/ddelta
289                  + exact sextupole feed-down             (Eqs. 99-101)
290
291        Parameters
292        ----------
293        energy, kicks_h, x_sex, xp_sex :
294            Feedback ingredients (horizontal), each either the numerical feedback
295            result or its analytical estimate from `feedback_ingredients`.
296        dRij_dE : (n_BPM, n_cor), optional
297            dR/ddelta in this plane; if None it is computed analytically with
298            `dRij_dEnergy` (base optics only).
299
300        Returns shape (n_CFD, n_BPM, n_cor).
301        """
302        if dRij_dE is None:
303            dRij_dE = self.dRij_dEnergy()
304
305        bpm, cor, cfd = self._private_element("bpm"), self._private_element(self.ind["cor"][self.dir]), self._private_element("CFD")
306
307        J = np.real(self.dRij_dqk_thick23(bpm, cor, cfd))          # K1 term
308        if self.dir_ind == 0:                                      # horizontal dispersion terms
309            J = J + np.real(self.dRij_dqk_thick23_disp(bpm, cor, cfd))
310            J = J + np.real(self.dRij_dbend_thick23_disp(bpm, cor, cfd))
311        J = J + self.dRij_dk_energy_term(bpm, cor, cfd, dRij_dE, energy)   # energy term
312        J = J + self.dRij_dCFD_sex_term(x_sex, xp_sex, kicks_h, energy)    # sextupole feed-down
313        return J

181    def _sex_dtheta(self, kicks_h):
182        """Map the horizontal feedback kicks (n_pert, n_cor_h) to the kick of the
183        corrector coil housed inside each sextupole (n_pert, n_sex), zero where the
184        sextupole has no corrector coil.  The feed-down orbit is horizontal, so the
185        relevant coils are always the horizontal correctors regardless of plane."""
186        hcm_col = {int(m): c for c, m in enumerate(self.ind["cor"]["h"])}

188        for s_i, l in enumerate(self.ind["sex"]):
189            c = hcm_col.get(int(l))
190            if c is not None:

194    def dRij_dCFD_sex_term(self, x_sex, xp_sex, kicks_h, energy):
195        """
196        Exact sextupole feed-down contribution to the CFD ORM Jacobian in this
197        plane (Eqs. 99-101 of ACDIV-2017-01), assembled from the feedback orbit
198        at the sextupoles.
199
200        The orbit that drives the feed-down is the *corrector-driven* part only:
201        the dispersion orbit eta*delta is subtracted because its feed-down is
202        already carried by the energy term dR/ddelta (with sextupoles ON this is
203        the chromaticity correction itself; keeping it in both double-counts).
204
205        Parameters
206        ----------
207        x_sex, xp_sex : (n_pert, n_sex)
208            Horizontal orbit and slope at the sextupole entrances per unit CFD
209            change (numerical feedback results or their analytical estimates).
210        kicks_h : (n_pert, n_cor_h)
211            Horizontal feedback kicks per unit CFD change.
212        energy : (n_pert,)
213            Energy shift dp/p per unit CFD change applied by the RF feedback.
214
215        Returns shape (n_pert, n_BPM, n_cor).
216        """
217        eta_s  = np.array([self.all_optics[2]["dispersion"][i][0] for i in self.ind["sex"]])
218        etap_s = np.array([self.all_optics[2]["dispersion"][i][1] for i in self.ind["sex"]])
219        dtheta = self._sex_dtheta(kicks_h)
220
221        i_fd = (x_sex  - np.outer(energy, eta_s ))[:, None, None, :]
222        j_fd = (xp_sex - np.outer(energy, etap_s))[:, None, None, :]
223        k_fd = dtheta[:, None, None, :]
224
225        bpm = self._private_element("bpm")
226        cor = self._private_element(self.ind["cor"][self.dir])
227        sex = self._private_element("sex")
228        bpm.broadcasters(1, 4)
229        cor.broadcasters(2, 4)
230        sex.broadcasters(3, 4)
231        # Ek is unused inside dRi_dk_sex_term (kept for signature symmetry) -> pass bpm.
232        return self.dRi_dk_sex_term(bpm, cor, bpm, sex, i_fd, j_fd, k_fd)
233
234    def feedback_ingredients(self):
235        """
236        Analytical estimates of the feedback quantities needed to assemble the CFD
237        ORM (and dispersion) Jacobians, all from the single base-optics call:
238
239          - energy : (n_CFD,)          energy shift dp/p per unit CFD change
240          - kicks_h: (n_CFD, n_cor_h)  horizontal feedback kicks per unit CFD change
241          - x_sex  : (n_CFD, n_sex)    horizontal orbit at the sextupole entrances
242          - xp_sex : (n_CFD, n_sex)    horizontal slope at the sextupole entrances
243
244        These reproduce the numerical Cor_SVD feedback results (energy via
245        `dCFD_denergy`, kicks via `dkicksdCFDk`, orbit via `dxldCFDk`, slope by a
246        finite difference of the orbit across the field-free drift upstream of each
247        sextupole — exact because the orbit is a straight line in a drift).
248
249        Must be called on a *horizontal* AnaORM (the Cor_SVD feedback is horizontal).
250        Returns a dict ready to pass to `dRij_dCFD`.
251        """
252        # energy shift per CFD
253        bpm, cor, cfd = self._private_element("bpm"), self._private_element(self.ind["cor"][self.dir]), self._private_element("CFD")
254        bpm.broadcasters(0, 2); cor.broadcasters(1, 2); cfd.broadcasters(1, 2)
255        energy = np.real(self.dCFD_denergy(bpm, cor, cfd))
256
257        # horizontal feedback kicks per CFD
258        bpm, cor, cfd = self._private_element("bpm"), self._private_element(self.ind["cor"][self.dir]), self._private_element("CFD")
259        bpm.broadcasters(0, 3); cor.broadcasters(1, 3); cfd.broadcasters(2, 3)
260        kicks_h = np.real(self.dkicksdCFDk(bpm, cor, cfd))
261
262        # orbit and slope at the sextupoles (slope via the upstream drift)
263        up, L_drift = [], []
264        for l in self.ind["sex"]:
265            j = int(l) - 1
266            while self.ring[j].Length == 0:
267                j -= 1
268            up.append(j)
269            L_drift.append(self.ring[j].Length)
270        L_drift = np.array(L_drift)
271
272        bpm, cor, cfd = self._private_element("bpm"), self._private_element(self.ind["cor"][self.dir]), self._private_element("CFD")
273        sex, sexup = self._private_element("sex"), self._private_element(up)
274        bpm.broadcasters(0, 4); cor.broadcasters(1, 4); cfd.broadcasters(2, 4)
275        sex.broadcasters(3, 4); sexup.broadcasters(3, 4)
276        x_sex = np.real(self.dxldCFDk(bpm, cor, cfd, sex))
277        x_up  = np.real(self.dxldCFDk(bpm, cor, cfd, sexup))
278        xp_sex = (x_sex - x_up) / L_drift
279
280        return {"energy": energy, "kicks_h": kicks_h, "x_sex": x_sex, "xp_sex": xp_sex}

282    def dRij_dCFD(self, energy, kicks_h, x_sex, xp_sex, dRij_dE=None):
283        """
284        Full ORM Jacobian dR/dK_CFD in this plane under active Cor_SVD orbit + RF
285        feedback:
286
287            dR/dK = quadrupole (K1) term
288                  + dispersion-term derivatives          (horizontal only)
289                  + energy term  delta_k * dR/ddelta
290                  + exact sextupole feed-down             (Eqs. 99-101)
291
292        Parameters
293        ----------
294        energy, kicks_h, x_sex, xp_sex :
295            Feedback ingredients (horizontal), each either the numerical feedback
296            result or its analytical estimate from `feedback_ingredients`.
297        dRij_dE : (n_BPM, n_cor), optional
298            dR/ddelta in this plane; if None it is computed analytically with
299            `dRij_dEnergy` (base optics only).
300
301        Returns shape (n_CFD, n_BPM, n_cor).
302        """
303        if dRij_dE is None:
304            dRij_dE = self.dRij_dEnergy()
305
306        bpm, cor, cfd = self._private_element("bpm"), self._private_element(self.ind["cor"][self.dir]), self._private_element("CFD")
307
308        J = np.real(self.dRij_dqk_thick23(bpm, cor, cfd))          # K1 term
309        if self.dir_ind == 0:                                      # horizontal dispersion terms
310            J = J + np.real(self.dRij_dqk_thick23_disp(bpm, cor, cfd))
311            J = J + np.real(self.dRij_dbend_thick23_disp(bpm, cor, cfd))
312        J = J + self.dRij_dk_energy_term(bpm, cor, cfd, dRij_dE, energy)   # energy term
313        J = J + self.dRij_dCFD_sex_term(x_sex, xp_sex, kicks_h, energy)    # sextupole feed-down
314        return J

102    def dRij_dEnergy(self, fringes=True):
103        """
104        Analytical derivative of the ORM with respect to the beam energy
105        deviation, dR/ddelta, from the base optics only (no RF-shifted rings).
106
107        Chain rule over the chromatic gradient sources:
108          - quadrupoles and CFDs: dK1/ddelta = -PolynomB[1] (the effective
109            gradient scales as K1/(1+delta)), thick formula dRij_dqk_thick23
110          - dipole edge focusing at the CFD faces (thin fringe quadrupoles,
111            K1L_edge = -h*tan(E) scaling as 1/(1+delta)), thin formula
112            dRij_dqk_thick2 with the local face optics
113          - sextupoles: feed-down from the dispersion orbit eta*ddelta,
114            integrated exactly along each sextupole (dRi_dk_sex_term with
115            i = eta, j = eta', k = 0)
116
117        Returns shape (n_BPM, n_cor).  Uses private Elements instances, so the
118        broadcasting state of self.bpm/self.cor/... is not modified.
119        """
120        bpm = Elements(self.ring, self.all_optics, self.ind["bpm"], self.dir_ind, self.sgn)
121        cor = Elements(self.ring, self.all_optics, self.ind["cor"][self.dir], self.dir_ind, self.sgn)
122        bpm.broadcasters(1, 3)
123        cor.broadcasters(2, 3)
124
125        out = 0.0
126        # thick gradients: quads and CFDs
127        for key in ("quad", "CFD"):
128            if key not in self.ind or len(self.ind[key]) == 0:
129                continue
130            el = Elements(self.ring, self.all_optics, self.ind[key], self.dir_ind, self.sgn)
131            el.broadcasters(0, 3)
132            w = -np.array([self.ring[i].PolynomB[1] for i in self.ind[key]])
133            out = out + self.dRij_dqk_thick23_wsum(bpm, cor, el, w)
134
135        # Thin fringe quadrupoles at the CFD faces, horizontal plane only:
136        # empirically the vertical edge-chromaticity contribution cancels in the
137        # AT edge model (adding it degrades the vertical dR/ddelta), while the
138        # horizontal one improves the agreement substantially.
139        # TODO: derive the vertical cancellation from the AT edge_fringe map.
140        if fringes and self.dir_ind == 0 and "CFD" in self.ind:
141            cfd_ind = np.asarray(self.ind["CFD"])
142            h = np.array([getattr(self.ring[i], "BendingAngle", 0.0) / self.ring[i].Length
143                          for i in cfd_ind])
144            for face_ind, w_face in [
145                (cfd_ind,     h * np.tan(np.array([getattr(self.ring[i], "EntranceAngle", 0.0) for i in cfd_ind]))),
146                (cfd_ind + 1, h * np.tan(np.array([getattr(self.ring[i], "ExitAngle",     0.0) for i in cfd_ind]))),
147            ]:
148                face = Elements(self.ring, self.all_optics, face_ind, self.dir_ind, self.sgn)
149                face.broadcasters(0, 3)
150                out = out + self.dRij_dqk_thick2_wsum(bpm, cor, face, w_face)
151
152        # sextupole feed-down from the dispersion orbit
153        if "sex" in self.ind and len(self.ind["sex"]) > 0:
154            bpm4 = Elements(self.ring, self.all_optics, self.ind["bpm"], self.dir_ind, self.sgn)
155            cor4 = Elements(self.ring, self.all_optics, self.ind["cor"][self.dir], self.dir_ind, self.sgn)
156            sex4 = Elements(self.ring, self.all_optics, self.ind["sex"], self.dir_ind, self.sgn)
157            bpm4.broadcasters(1, 4)
